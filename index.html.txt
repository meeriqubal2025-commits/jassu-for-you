    <script>
        const steps = [
            {
                title: "🎁 Step 1",
                text: "Here is a small virtual gift for someone very special...",
                button: "Open Gift ❤️",
                type: "text"
            },
            {
                title: "✨ Step 2",
                text: "Do you want to see the most gorgeous girl in the world?",
                button: "Yes, Show Me! 😍",
                type: "text"
            },
            {
                title: "🔤 Step 3",
                text: "Her beautiful name starts with the letter **'J'**...",
                button: "Aha! Go Ahead 💫",
                type: "text"
            },
            {
                title: "👀 Step 4",
                text: "Are you ready to witness true beauty?",
                button: "Ready! 🔥",
                type: "text"
            },
            {
                title: "⚠️ Step 5 (Warning)",
                text: "Warning: You might go blind after seeing too much beauty! Proceed at your own risk. 😉",
                button: "I'll Take The Risk! 🙈",
                type: "text"
            },
            {
                title: "My Everything ❤️",
                image: "photo.jpg",
                text: "You are the brightest light in my life. Thank you for making every moment magical. I love you endlessly! ✨",
                button: "Forever Yours 💖",
                type: "final"
            }
        ];

        let currentStep = 0;

        function updateStep() {
            const box = document.getElementById("content-box");
            const step = steps[currentStep];

            if (step.type === "text") {
                box.innerHTML = `
                    <h2>${step.title}</h2>
                    <p>${step.text}</p>
                    <button class="btn" onclick="nextStep()">${step.button}</button>
                `;
            } else if (step.type === "final") {
                box.innerHTML = `
                    <img src="${step.image}" alt="Gorgeous" class="photo">
                    <h2>${step.title}</h2>
                    <p>${step.text}</p>
                    <button class="btn" onclick="restart()">Replay ❤️</button>
                `;
            }
        }

        function nextStep() {
            currentStep++;
            if (currentStep < steps.length) {
                updateStep();
            }
        }

        function restart() {
            currentStep = 0;
            updateStep();
        }

        updateStep();
    </script>
